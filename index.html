  c.cohort_name,
  COUNT(s.student_id) as enrolled,
  ROUND(AVG(s.salary_hike), 1) as avg_hike
FROM cohorts c
JOIN students s ON s.cohort_id = c.id
WHERE s.placement_status = 'Placed'
GROUP BY c.cohort_name;
